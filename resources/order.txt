select round(((new_price - old_price)/old_price * 100), 2) as diff_per1, b.*, c.*, round(((b.new_price - c.today)/ c.today * 100)::numeric, 0) as diff_per2 from hold_tickers a inner join reasonable_prices b on a.ticker = b.ticker
left outer join yahoo_price c on a.ticker = c.ticker order by diff_per2 desc;

--WithInvestingComと、yahoo_priceのdiff_per見る（0除算エラー回避バージョン）
SELECT ROUND(((new_price - old_price)/old_price * 100), 2) AS diff_per1, b.*, c.*, CASE WHEN c.today <> 0 THEN ROUND(((b.new_price - c.today) / c.today * 100)::numeric, 0) ELSE NULL END AS diff_per2 FROM hold_tickers a
INNER JOIN reasonable_prices b ON a.ticker = b.ticker LEFT OUTER JOIN yahoo_price c ON a.ticker = c.ticker where c.create_date = CURRENT_DATE ORDER BY diff_per2 DESC;

select * from rating_report a
	left join rating_report_price b on a.ticker = b.ticker left join closing_date cd on a.ticker = cd.ticker left join reasonable_prices rp on a.ticker = rp.ticker
	where a.new in ('買い', 'Ｂｕｙ', '１', 'Ａ', 'アウトパフォーム', 'オーバーウエート') and a.create_date != CURRENT_DATE and b.create_date != CURRENT_DATE and (b.today < rp.new_price or rp.new_price is null)
	and a.create_date >= (CURRENT_DATE - INTERVAL '7 days') and  b.create_date > (CURRENT_DATE - INTERVAL '7 days') and a.diff_per > 0 order by a.diff_per desc;
