select (a.to_price - b.today) as price_diff, round((a.to_price - b.today)/ b.today * 100) as price_diff_per, * from rating_report a 
	left join rating_report_price b on a.ticker = b.ticker
	left join fiscal_dates c on a.ticker = c.ticker
	where a.create_date = '2024-02-14' and b.create_date = '2024-02-14' order by a.diff_per desc;
--	where a.create_date = '2024-02-14' order by a.diff_per desc;

select (a.to_price - b.today) as price_diff, round((a.to_price - b.today)/ b.today * 100) as price_diff_per, * from rating_report a
	left join rating_report_price b on a.ticker = b.ticker
	left join fiscal_dates c on a.ticker = c.ticker
	where a.new in ('買い', 'Ｂｕｙ', '１', 'Ａ', 'アウトパフォーム') and a.create_date = '2024-02-15' and b.create_date = '2024-02-15' order by a.diff_per desc;
